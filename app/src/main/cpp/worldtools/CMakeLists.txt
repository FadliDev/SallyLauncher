cmake_minimum_required(VERSION 3.22)
project(sallyworld LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF)
set(SNAPPY_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SNAPPY_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(SNAPPY_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/snappy EXCLUDE_FROM_ALL)
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZSTD_MULTITHREAD_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/zstd/build/cmake EXCLUDE_FROM_ALL)
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
set(LEVELDB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LEVELDB_INSTALL OFF CACHE BOOL "" FORCE)
set(LEVELDB_TCMALLOC OFF CACHE BOOL "" FORCE)
set(LEVELDB_SNAPPY ON CACHE BOOL "" FORCE)
set(LEVELDB_ZSTD ON CACHE BOOL "" FORCE)
set(LEVELDB_ZLIB ON CACHE BOOL "" FORCE)
add_subdirectory(vendor/leveldb EXCLUDE_FROM_ALL)
file(GLOB_RECURSE NBT_SOURCES CONFIGURE_DEPENDS vendor/nbt/src/nbt/*.cpp)
add_library(sallyworld SHARED
    WorldJni.cpp
    ${NBT_SOURCES}
    vendor/binarystream/src/binarystream/BinaryStream.cpp
    vendor/binarystream/src/binarystream/ReadOnlyBinaryStream.cpp)
target_include_directories(sallyworld PRIVATE vendor/nbt/include vendor/nbt/src vendor/binarystream/include)
target_compile_features(sallyworld PRIVATE cxx_std_23)
target_compile_options(sallyworld PRIVATE -fexceptions -frtti -fvisibility=hidden -fvisibility-inlines-hidden)
target_link_libraries(sallyworld PRIVATE leveldb z)
target_link_options(sallyworld PRIVATE -Wl,--exclude-libs,ALL -Wl,-Bsymbolic -Wl,-z,max-page-size=16384)
